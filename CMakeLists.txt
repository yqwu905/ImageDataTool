cmake_minimum_required(VERSION 3.16)
project(ImageCompare VERSION 1.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(WIN32)
    enable_language(RC)
    add_executable(ImageCompare WIN32 src/main.cpp assets/app.rc)
    target_include_directories(ImageCompare PRIVATE assets)
    target_link_libraries(ImageCompare PRIVATE gdiplus ole32 shell32 uuid dwmapi comctl32 gdi32 windowscodecs)
    if(MSVC)
        set_property(TARGET ImageCompare PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
        target_compile_options(ImageCompare PRIVATE /W4 /utf-8)
        target_link_options(ImageCompare PRIVATE /OPT:REF /OPT:ICF /MANIFEST:NO)
    else()
        target_compile_options(ImageCompare PRIVATE -Wall -Wextra -ffunction-sections -fdata-sections)
        target_link_options(ImageCompare PRIVATE -municode -static -static-libgcc -static-libstdc++ -Wl,--gc-sections -s)
    endif()
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(core_tests tests/core_tests.cpp)
    add_test(NAME core COMMAND core_tests)
    set_tests_properties(core PROPERTIES TIMEOUT 60)
    if(WIN32)
        add_executable(windows_tests tests/windows_tests.cpp)
        target_link_libraries(windows_tests PRIVATE gdiplus ole32 shell32 uuid dwmapi comctl32 gdi32 windowscodecs)
        if(MSVC)
            target_compile_options(core_tests PRIVATE /utf-8)
            target_compile_options(windows_tests PRIVATE /utf-8)
            set_property(TARGET core_tests windows_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
        else()
            target_link_options(windows_tests PRIVATE -municode -static)
            target_link_options(core_tests PRIVATE -static)
        endif()
        add_test(NAME windows_integration COMMAND windows_tests "${CMAKE_BINARY_DIR}/test-fixtures" "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
        set_tests_properties(windows_integration PROPERTIES TIMEOUT 90)
    endif()
endif()

option(IMAGECOMPARE_BUILD_BENCHMARKS "Build reproducible viewer performance harness" OFF)
set(IMAGECOMPARE_BASELINE_DIR "" CACHE PATH "Optional v1.0.0 source tree for before/after measurements")
if(WIN32 AND IMAGECOMPARE_BUILD_BENCHMARKS)
    add_executable(performance_benchmark tests/performance_benchmark.cpp)
    set(benchmark_targets performance_benchmark)
    if(IMAGECOMPARE_BASELINE_DIR)
        add_executable(performance_baseline tests/performance_benchmark.cpp)
        target_compile_definitions(performance_baseline PRIVATE IC_APP_SOURCE="${IMAGECOMPARE_BASELINE_DIR}/src/main.cpp")
        list(APPEND benchmark_targets performance_baseline)
    endif()
    foreach(target IN LISTS benchmark_targets)
        target_link_libraries(${target} PRIVATE gdiplus ole32 shell32 uuid dwmapi comctl32 gdi32 windowscodecs)
        if(MSVC)
            target_compile_options(${target} PRIVATE /utf-8)
            set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
        else()
            target_link_options(${target} PRIVATE -municode -static)
        endif()
    endforeach()
endif()
